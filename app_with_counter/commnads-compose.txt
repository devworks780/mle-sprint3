# комбинация предыдущих команд: 
# мы пересобираем образы и сразу запускаем сервисы

docker compose up <имя сервиса> --build
# docker compose up counter-app --build

docker compose stop <имя сервиса>
# остановит все контейнеры, ассоциированные с сервисом

docker compose start <имя сервиса>
# запустит все контейнеры, ассоциированные с сервисом
# если они уже запущены, то ничего не сделает

docker compose restart <имя сервиса>
# перезапустит контейнеры, ассоциированные с сервисом
# в отличие от команды start выполнит перезапуск в любом случае 

docker compose down <имя сервиса>
